cmake_minimum_required(VERSION 3.16)
project(HPC_Style_For_AI LANGUAGES CXX)
option(HPC_STYLE_OPENMP "Test the OpenMP execution adapter" OFF)
add_executable(owner_view examples/owner_view.cpp)
target_compile_features(owner_view PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(owner_view PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(HPC_STYLE_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_link_libraries(owner_view PRIVATE OpenMP::OpenMP_CXX)
endif()
enable_testing()
add_test(NAME owner_view_correctness COMMAND owner_view)
set_tests_properties(owner_view_correctness PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=4")

option(HPC_STYLE_KOKKOS "Build optional host Kokkos contract probes (no downloads)" OFF)
if(HPC_STYLE_KOKKOS)
    find_package(Kokkos 5.3 CONFIG REQUIRED)
    add_executable(kokkos_contracts examples/kokkos_contracts.cpp)
    target_link_libraries(kokkos_contracts PRIVATE Kokkos::kokkos)
    target_compile_features(kokkos_contracts PRIVATE cxx_std_20)
    if(Kokkos_ENABLE_OPENMP)
        add_test(NAME kokkos_contracts_1thread COMMAND kokkos_contracts --kokkos-num-threads=1)
        add_test(NAME kokkos_contracts_4threads COMMAND kokkos_contracts --kokkos-num-threads=4)
        set_tests_properties(kokkos_contracts_1thread PROPERTIES ENVIRONMENT "KOKKOS_NUM_THREADS=1;OMP_NUM_THREADS=1;HPC_STYLE_EXPECT_THREADS=1;OMP_PROC_BIND=false")
        set_tests_properties(kokkos_contracts_4threads PROPERTIES ENVIRONMENT "KOKKOS_NUM_THREADS=4;OMP_NUM_THREADS=4;HPC_STYLE_EXPECT_THREADS=4;OMP_PROC_BIND=false")
    else()
        add_test(NAME kokkos_contracts_serial COMMAND kokkos_contracts)
    endif()
endif()
